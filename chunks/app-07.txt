function dailyAudioFingerprint(left,right){
  const n=Math.min(left.length,right.length);if(!n)return {v:[]};
  const stride=Math.max(1,Math.floor(n/180000)),segments=16,segVals=new Array(segments).fill(0),segN=new Array(segments).fill(0);
  let sum=0,peak=0,diff=0,side=0,mid=0,zc=0,count=0,prev=(left[0]+right[0])*.5;
  for(let i=0;i<n;i+=stride){
    const l=left[i],r=right[i],m=(l+r)*.5,x=Math.abs(m);sum+=m*m;peak=Math.max(peak,x);
    const d=m-prev;diff+=d*d;if((m>=0)!=(prev>=0))zc++;prev=m;side+=(l-r)*(l-r);mid+=(l+r)*(l+r);count++;
    const s=Math.min(segments-1,Math.floor((i/n)*segments));segVals[s]+=m*m;segN[s]++;
  }
  const rms=Math.sqrt(sum/Math.max(1,count)),crest=peak/Math.max(.000001,rms),zcr=zc/Math.max(1,count),brightness=Math.sqrt(diff/Math.max(1,count))/Math.max(.000001,rms);
  const width=Math.sqrt(side/Math.max(1,count))/Math.max(.000001,Math.sqrt(mid/Math.max(1,count)));
  const env=segVals.map((x,i)=>Math.sqrt(x/Math.max(1,segN[i]))),mean=env.reduce((a,b)=>a+b,0)/env.length;
  const envNorm=env.map(x=>clamp(x/Math.max(.000001,mean*2),0,1)),envStd=Math.sqrt(env.reduce((a,x)=>a+(x-mean)*(x-mean),0)/env.length)/Math.max(.000001,mean);
  const silence=env.filter(x=>x<Math.max(.0025,mean*.18)).length/env.length;
  return {v:[
    clamp((Math.log10(Math.max(1e-5,rms))+5)/5,0,1),clamp(crest/10,0,1),clamp(zcr*12,0,1),
    clamp(brightness/3,0,1),clamp(width/2,0,1),clamp(envStd/1.5,0,1),silence,...envNorm
  ]};
}

function resampleLinearStereo(left,right,fromRate,toRate=44100){
  if(fromRate===toRate)return {left,right};
  const outLen=Math.max(1,Math.round(left.length*toRate/fromRate)),l=new Float32Array(outLen),r=new Float32Array(outLen),ratio=fromRate/toRate;
  for(let i=0;i<outLen;i++){const p=i*ratio,j=Math.floor(p),f=p-j,j2=Math.min(left.length-1,j+1);l[i]=left[j]*(1-f)+left[j2]*f;r[i]=right[j]*(1-f)+right[j2]*f}
  return {left:l,right:r};
}

function estimateIntegratedLoudness(left,right,sampleRate){
  const n=Math.min(left.length,right.length),block=Math.max(1,Math.round(sampleRate*.4)),hop=Math.max(1,Math.round(sampleRate*.1));
  const energies=[];
  for(let start=0;start+block<=n;start+=hop){
    let sum=0;for(let i=start;i<start+block;i++){const l=left[i],r=right[i];sum+=l*l+r*r}
    const e=sum/(block*2),lu=-.691+10*Math.log10(Math.max(1e-12,e));if(lu>-70)energies.push({e,lu});
  }
  if(!energies.length)return -70;
  let mean=energies.reduce((a,b)=>a+b.e,0)/energies.length,pre=-.691+10*Math.log10(Math.max(1e-12,mean)),gate=Math.max(-70,pre-10);
  const gated=energies.filter(x=>x.lu>=gate);if(!gated.length)return pre;
  mean=gated.reduce((a,b)=>a+b.e,0)/gated.length;
  return -.691+10*Math.log10(Math.max(1e-12,mean));
}

  const length=Math.min(left.length,right.length),offline=new OfflineAudioContext(2,length,sampleRate),buf=offline.createBuffer(2,length,sampleRate);
  buf.copyToChannel(left.subarray(0,length),0);buf.copyToChannel(right.subarray(0,length),1);
  const src=offline.createBufferSource(),g=offline.createGain(),lim=offline.createDynamicsCompressor();
  src.buffer=buf;g.gain.value=gain;
  lim.threshold.value=-1.1;lim.knee.value=0;lim.ratio.value=20;lim.attack.value=.002;lim.release.value=.09;
  src.connect(g).connect(lim).connect(offline.destination);src.start();
  const rendered=await offline.startRendering(),l=new Float32Array(rendered.getChannelData(0)),r=new Float32Array(rendered.getChannelData(1));
  let peak=0;for(let i=0;i<length;i++)peak=Math.max(peak,Math.abs(l[i]),Math.abs(r[i]));
  const trim=peak>ceiling?ceiling/peak:1;if(trim<1)for(let i=0;i<length;i++){l[i]*=trim;r[i]*=trim}
  const after=estimateIntegratedLoudness(l,r,sampleRate);
  return {left:l,right:r,mastering:{targetLUFS:target,ceilingDbTP:ceilingDb,inputLUFS:before,outputLUFS:after,appliedGainDb:gainDb,peakDbFS:20*Math.log10(Math.max(1e-9,peak*trim))}};
}
