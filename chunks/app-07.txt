function dailyDateStamp(){
  const d=new Date(),p=n=>String(n).padStart(2,'0');
  return d.getFullYear()+'-'+p(d.getMonth()+1)+'-'+p(d.getDate());
}

function concatFloat32(parts,total){
  const out=new Float32Array(total);let off=0;
  for(const p of parts){out.set(p,off);off+=p.length}
  return out;
}
function writeAscii(view,off,s){for(let i=0;i<s.length;i++)view.setUint8(off+i,s.charCodeAt(i))}
function encodeFloat32Wav(left,right,sampleRate){
  const frames=Math.min(left.length,right.length),channels=2,bits=32,bytesPerSample=4;
  const dataBytes=frames*channels*bytesPerSample;
  const ab=new ArrayBuffer(44+dataBytes),v=new DataView(ab);
  writeAscii(v,0,'RIFF');v.setUint32(4,36+dataBytes,true);writeAscii(v,8,'WAVE');
  writeAscii(v,12,'fmt ');v.setUint32(16,16,true);v.setUint16(20,3,true);
  v.setUint16(22,channels,true);v.setUint32(24,sampleRate,true);
  v.setUint32(28,sampleRate*channels*bytesPerSample,true);
  v.setUint16(32,channels*bytesPerSample,true);v.setUint16(34,bits,true);
  writeAscii(v,36,'data');v.setUint32(40,dataBytes,true);
  let o=44;
  for(let i=0;i<frames;i++){v.setFloat32(o,left[i],true);o+=4;v.setFloat32(o,right[i],true);o+=4}
  return new Blob([ab],{type:'audio/wav'});
}
function beginDailyPcmCapture(){
  if(!ctx||!master)throw new Error('Audio engine is not ready');
  const node=ctx.createScriptProcessor(4096,2,2),sink=ctx.createGain();
  sink.gain.value=0;
  const left=[],right=[];let total=0;
  node.onaudioprocess=e=>{
    if(!dailyCapture||dailyCapture.node!==node)return;
    const l=e.inputBuffer.getChannelData(0),r=e.inputBuffer.numberOfChannels>1?e.inputBuffer.getChannelData(1):l;
    left.push(new Float32Array(l));right.push(new Float32Array(r));total+=l.length;
  };
  master.connect(node);node.connect(sink);sink.connect(ctx.destination);
  dailyCapture={node,sink,left,right,get total(){return total}};
}
function endDailyPcmCapture(){
  const cap=dailyCapture;if(!cap)return null;dailyCapture=null;
  try{cap.node.onaudioprocess=null;master.disconnect(cap.node);cap.node.disconnect();cap.sink.disconnect()}catch(e){}
  const l=concatFloat32(cap.left,cap.total),r=concatFloat32(cap.right,cap.total);
  return encodeFloat32Wav(l,r,ctx.sampleRate);
}

async function dailyRecordOne(label,index){
  if(dailyCancelRequested)return false;
  if(!running)await start();
  if(!ctx||!master)throw new Error('Audio engine is unavailable');
  if(label==='A'&&rhythmOn)toggleRhythm();

  if(label!=='A')newGroove(true);
  allRandomMorph();
  const duration=Math.round(120+rand()*60);
  beginDailyPcmCapture();
  const started=Date.now(),status=$('#dailyStatus');
  while((Date.now()-started)<duration*1000&&!dailyCancelRequested){
    const elapsed=Math.floor((Date.now()-started)/1000),left=Math.max(0,duration-elapsed);
    if(status)status.textContent='TRACK '+label+' / 3 — WAV '+Math.round(ctx.sampleRate/1000)+'kHz / 32-bit float · '+Math.floor(left/60)+':'+String(left%60).padStart(2,'0')+' remaining';
    await new Promise(r=>setTimeout(r,1000));
  }
  const blob=endDailyPcmCapture();
  if(dailyCancelRequested||!blob)return false;
  dailyRenderResult(label,blob,dailySnapshot(label,duration));
  return true;
}

function randomValue(c,range=null){let lo=range?.min??c.rmin??c.min,hi=range?.max??c.rmax??c.max;if(hi<lo)[lo,hi]=[hi,lo];if(c.scale==='log'){lo=Math.max(c.min,lo);hi=Math.max(lo,hi);const a=Math.log(lo),b=Math.log(hi);return Math.exp(a+rand()*(b-a))}return lo+rand()*(hi-lo)}

// A true continuous bounded random walk. Every sound-making control has its own
// independent velocity. Directions bend gradually and reflect at user-defined MIN/MAX.
const walkAgents={ambient:{},rhythm:{},mixer:{}};let walkLast=performance.now(),walkReadoutAt=0;
function makeWalkAgent(){return {v:(Math.random()<.5?-1:1)*(.003+Math.random()*.009),targetV:0,turnIn:.25+Math.random()*1.8,personality:.65+Math.random()*.9}}
function resetWalkAgents(){walkAgents.ambient={};walkAgents.rhythm={};walkAgents.mixer={};for(const id of Object.keys(configs))walkAgents.ambient[id]=makeWalkAgent();for(const id of Object.keys(rhythmConfigs))walkAgents.rhythm[id]=makeWalkAgent();for(const id of Object.keys(mixerConfigs))walkAgents.mixer[id]=makeWalkAgent();walkLast=performance.now()}
function walkParameter(target,id,c,range,agent,dt){if(!range||range.max<=range.min)return false;let n=toNorm(target[id],c),lo=toNorm(range.min,c),hi=toNorm(range.max,c);if(hi<lo)[lo,hi]=[hi,lo];n=clamp(n,lo,hi);agent.turnIn-=dt;const speed=meta.walkSpeed,depth=meta.depth;if(agent.turnIn<=0||!Number.isFinite(agent.targetV)){const sign=Math.random()<.5?-1:1;const mag=(.0035+Math.random()*.0135)*speed*agent.personality*(.55+.65*depth);agent.targetV=sign*mag;agent.turnIn=.35+Math.random()*(2.8/(.45+speed*.55))}const bend=1-Math.exp(-dt*(.45+speed*.7));agent.v+=(agent.targetV-agent.v)*bend;const minV=.0015*speed*agent.personality;if(Math.abs(agent.v)<minV)agent.v=(agent.v<0?-1:1)*minV;n+=agent.v*dt;if(n<=lo){n=lo+(lo-n);agent.v=Math.abs(agent.v);agent.targetV=Math.abs(agent.targetV||agent.v);agent.turnIn=.2+Math.random()*.9}if(n>=hi){n=hi-(n-hi);agent.v=-Math.abs(agent.v);agent.targetV=-Math.abs(agent.targetV||agent.v);agent.turnIn=.2+Math.random()*.9}n=clamp(n,lo,hi);target[id]=fromNorm(n,c);return true}
function updateRandomWalk(now){const dt=Math.min(.08,Math.max(0,(now-walkLast)/1000));walkLast=now;if(evolving&&dt>0){for(const [id,c] of Object.entries(configs)){if(locked.has(id))continue;walkParameter(state,id,c,randomRanges[id],walkAgents.ambient[id]||(walkAgents.ambient[id]=makeWalkAgent()),dt);syncControl(id,state,c)}for(const [id,c] of Object.entries(rhythmConfigs)){if(rhythmLocked.has(id))continue;walkParameter(rhythmState,id,c,rhythmRandomRanges[id],walkAgents.rhythm[id]||(walkAgents.rhythm[id]=makeWalkAgent()),dt);syncRhythmControl(id)}for(const [id,c] of Object.entries(mixerConfigs)){if(mixerLocked.has(id))continue;walkParameter(mixerState,id,c,mixerRandomRanges[id],walkAgents.mixer[id]||(walkAgents.mixer[id]=makeWalkAgent()),dt);syncMixerControl(id)}grooveBlendAmount=clamp(rhythmState.grooveMix,0,1);if(now-walkReadoutAt>180){updateGrooveReadout();walkReadoutAt=now}applyGlobals()}requestAnimationFrame(updateRandomWalk)}
