
// Sine particles: independent phrases, gently influenced by the current groove.
const sineChords={sus2:[0,2,7],sus4:[0,5,7],'7sus4':[0,5,7,10],'9sus4':[0,5,7,10,14],'add9(no3)':[0,7,14],'quartal':[0,5,10]};
const sineRoots=['C','C♯','D','E♭','E','F','F♯','G','A♭','A','B♭','B'];
const sineDefaults={enabled:true,level:.5,decay:.22,density:.62,octave:4,range:2,link:.4,single:45,arp:40,chord:15,bars:2,root:0,direction:'updown',pool:['sus2','sus4','7sus4','9sus4']};
let sineState=JSON.parse(JSON.stringify(sineDefaults)),sineBus=null,sineNext=0,sineStep=0,sineBeat=0,sineChangeAt=0,sinePhraseAt=0,sineQuality='sus2',sineMode='single',sineArp=0,sineEvents=0;
const sineVoices=new Set();
const sineKnobs={level:['音量',0,1,.01],decay:['音の長さ',.045,1.2,.005],density:['鳴る密度',0,1,.01],octave:['基準オクターブ',2,6,1],range:['音域（オクターブ）',1,3,1],link:['リズムとの連動度',0,1,.01],single:['単音の割合',0,100,1],arp:['アルペジオの割合',0,100,1],chord:['和音の割合',0,100,1],bars:['コード変更間隔（小節）',1,16,1]};
function sineValue(id){const v=sineState[id];return id==='decay'?Math.round(v*1000)+' ms':['level','density','link'].includes(id)?Math.round(v*100)+'%':['single','arp','chord'].includes(id)?v+'':v+''}
function syncSine(){
  const b=$('#sineToggle');b.textContent=sineState.enabled?'SYNTH ON':'SYNTH OFF';b.classList.toggle('on',sineState.enabled);b.setAttribute('aria-pressed',String(sineState.enabled));
  for(const id of Object.keys(sineKnobs)){$('#sine_'+id).value=sineState[id];$('#sineValue_'+id).textContent=sineValue(id)}
  $('#sineNow').textContent=sineRoots[sineState.root]+' '+sineQuality;
  $('#sineRoot').value=sineState.root;$('#sineDirection').value=sineState.direction;
  document.querySelectorAll('[data-sine-chord]').forEach(el=>el.checked=sineState.pool.includes(el.dataset.sineChord));
  if(sineBus&&ctx)sineBus.gain.setTargetAtTime(sineState.enabled?sineState.level:0,ctx.currentTime,.02);
}
function saveSine(){try{localStorage.setItem('karesansui-variant01-sine-v1',JSON.stringify(sineState))}catch(e){}}
function restoreSine(value){
  if(!value||typeof value!=='object')return;
  for(const [id,c] of Object.entries(sineKnobs)){const v=Number(value[id]);if(Number.isFinite(v))sineState[id]=Math.round(clamp(v,c[1],c[2])/c[3])*c[3]}
  if(typeof value.enabled==='boolean')sineState.enabled=value.enabled;
  if(Number.isInteger(value.root)&&value.root>=0&&value.root<12)sineState.root=value.root;
  if(['up','down','updown','random'].includes(value.direction))sineState.direction=value.direction;
  if(Array.isArray(value.pool)){const p=[...new Set(value.pool.filter(k=>Object.hasOwn(sineChords,k)))];if(p.length)sineState.pool=p}
  resetSine();syncSine();saveSine();
}
function resetSine(){
  for(const v of sineVoices){try{v.g.gain.cancelScheduledValues(ctx.currentTime);v.g.gain.setTargetAtTime(.00001,ctx.currentTime,.006);v.o.stop(ctx.currentTime+.04)}catch(e){}}
  sineNext=ctx?ctx.currentTime+.04:0;sineStep=0;sineBeat=0;sineChangeAt=0;sinePhraseAt=0;sineArp=0;
}
function chooseSineChord(){
  const others=sineState.pool.filter(k=>k!==sineQuality);sineQuality=(others.length?others:sineState.pool)[Math.floor(rand()*(others.length||sineState.pool.length))];sineArp=0;
  $('#sineNow').textContent=sineRoots[sineState.root]+' '+sineQuality;
}
function sineNotes(){const out=[];for(let oct=0;oct<sineState.range;oct++)for(const interval of sineChords[sineQuality])out.push(12*(sineState.octave+1)+sineState.root+interval+12*oct);return [...new Set(out)].sort((a,b)=>a-b)}
function sineNote(midi,t,velocity,count){
  if(sineVoices.size>=32)return;
  if(!sineBus){sineBus=ctx.createGain();sineBus.gain.value=sineState.level;sineBus.connect(compressor)}
  const o=ctx.createOscillator(),g=ctx.createGain(),p=ctx.createStereoPanner();o.type='sine';o.frequency.value=440*Math.pow(2,(midi-69)/12);p.pan.value=(rand()-.5)*.6;
  const duration=sineState.decay,amp=.22*velocity/Math.sqrt(count);g.gain.setValueAtTime(0,t);g.gain.linearRampToValueAtTime(amp,t+.004);g.gain.exponentialRampToValueAtTime(.00001,t+duration);g.gain.linearRampToValueAtTime(0,t+duration+.012);
  o.connect(g).connect(p).connect(sineBus);const voice={o,g,p};sineVoices.add(voice);o.onended=()=>{o.disconnect();g.disconnect();p.disconnect();sineVoices.delete(voice)};o.start(t);o.stop(t+duration+.02);
}
function sineScheduler(now,horizon){
  if(!sineState.enabled){sineNext=now+.04;return}
  if(sineNext<now-.1)sineNext=now+.01;
  const beat=60/state.tempo,link=sineState.link,genre=grooveFocus==='auto'?groovePrimary:grooveFocus;
  // Eighth-note pulse probabilities. Blend primary and secondary groove identities.
  const profiles={house:[1,.35,.7,.4,1,.35,.7,.5],techno:[1,.75,.8,.75,1,.75,.8,.75],jazz:[.8,.35,.5,1,.45,.7,1,.4],rnb:[.9,.2,.55,.7,.7,.3,.85,.45],hiphop:[1,.2,.55,.65,.8,.25,.8,.35],reggae:[.2,1,.25,.85,.2,1,.3,.85]};
  let guard=0;
  while(sineNext<horizon&&guard++<32){
    if(sineBeat>=sineChangeAt){chooseSineChord();sineChangeAt=sineBeat+4*sineState.bars}
    if(sineBeat>=sinePhraseAt){const sum=sineState.single+sineState.arp+sineState.chord,r=rand()*sum;sineMode=r<sineState.single?'single':r<sineState.single+sineState.arp?'arp':'chord';sinePhraseAt=sineBeat+4;sineArp=0}
    const ix=sineStep%8,a=(profiles[genre]||profiles.house)[ix],b=(profiles[grooveSecondary]||profiles.house)[ix],profile=a*(1-grooveBlendAmount)+b*grooveBlendAmount;
    const probability=sineState.density*((1-link)+link*profile);
    if(sineState.single+sineState.arp+sineState.chord>0&&rand()<probability){
      const notes=sineNotes();let chosen;
      if(sineMode==='chord')chosen=sineChords[sineQuality].map(n=>12*(sineState.octave+1)+sineState.root+n);
      else if(sineMode==='single'||sineState.direction==='random')chosen=[notes[Math.floor(rand()*notes.length)]];
      else {let i=sineArp++%notes.length;if(sineState.direction==='down')i=notes.length-1-i;if(sineState.direction==='updown'){const cycle=(sineArp-1)%Math.max(1,notes.length*2-2);i=cycle<notes.length?cycle:notes.length*2-2-cycle}chosen=[notes[i]]}
      const swing=sineStep%2?beat*.18*link*(genre==='jazz'?1:rhythmState.swing):0;
      const when=Math.max(now+.006,sineNext+swing+(rand()-.5)*beat*.05*(1-link));
      chosen.forEach(n=>sineNote(n,when,.65+rand()*.35,chosen.length));sineEvents++;
      $('#sineMode').textContent=({single:'SINGLE',arp:'ARPEGGIO',chord:'CHORD'})[sineMode]+' · '+sineEvents;
    }
    sineNext+=beat*.5*(1+(rand()-.5)*.65*(1-link));sineBeat+=.5;sineStep++;
  }
}
const sinePanel=document.createElement('section');sinePanel.className='sinePanel';sinePanel.id='sinePanel';sinePanel.setAttribute('aria-label','Sine synth laboratory');
sinePanel.innerHTML=`<div class="sineHeading"><div><div class="sectionTitle">SINE PARTICLES / TEST LAB</div><p>短い音の粒と、曖昧な和音。サンプルの隙間にシンセを重ねる。</p></div><button id="sineToggle" class="on">SYNTH ON</button></div><div class="sineSelectors"><label>ルート音 <select id="sineRoot">${sineRoots.map((n,i)=>`<option value="${i}">${n}</option>`).join('')}</select></label><label>アルペジオ <select id="sineDirection"><option value="up">上昇</option><option value="down">下降</option><option value="updown" selected>往復</option><option value="random">ランダム</option></select></label><button id="sineNewChord">NEXT CHORD ↝</button></div><fieldset class="sinePool"><legend>コード候補（複数選択）</legend>${Object.keys(sineChords).map(k=>`<label><input type="checkbox" data-sine-chord="${k}"> ${k}</label>`).join('')}</fieldset><div class="sineControls">${Object.entries(sineKnobs).map(([id,c])=>`<div><label class="sineControlLabel" for="sine_${id}">${c[0]} <output id="sineValue_${id}" for="sine_${id}"></output></label><input id="sine_${id}" type="range" min="${c[1]}" max="${c[2]}" step="${c[3]}"></div>`).join('')}</div><div class="sineReadout"><strong id="sineNow">—</strong><span id="sineMode">READY</span></div><p class="sineHint">上部のPLAYで再生。割合は相対的な重みです（すべて0で休止）。約1小節ごとに鳴り方を選びます。連動度0%は自由な間隔、100%はテンポとジャンルの傾向に沿います。RHYTHM OFFでも再生できます。設定は自動保存され、SAVE 1にも含まれます。</p>`;
