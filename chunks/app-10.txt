
// Two tactile XY surfaces; each has independent HOLD / bounded DRIFT.
const sinePadSpecs={grain:{title:'粒 / PARTICLES',x:'density',y:'decay',left:'まばら',right:'密集',bottom:'短い粒',top:'長い音'},space:{title:'空間 / SPACE',x:'reverb',y:'delay',left:'ドライ',right:'深い余韻',bottom:'反復少なめ',top:'反復多め'}};
const sinePadModels={};
function sinePadNorm(id){const c=sineKnobs[id];return clamp((sineState[id]-c[1])/(c[2]-c[1]),0,1)}
function sinePadApply(model,x,y){
  model.x=clamp(x,0,1);model.y=clamp(y,0,1);
  for(const [id,n] of [[model.spec.x,model.x],[model.spec.y,model.y]]){const c=sineKnobs[id];sineState[id]=c[1]+n*(c[2]-c[1]);$('#sine_'+id).value=sineState[id];$('#sineValue_'+id).textContent=sineValue(id)}
  applySineEffects();sinePadRender(model);
}
function sinePadRender(model){
  const {name,x,y}=model,mode=sineState[name+'Mode'];
  const dot=$('#xyDot_'+name);dot.style.left=(x*100)+'%';dot.style.top=((1-y)*100)+'%';
  const text=sineKnobs[model.spec.x][0]+' '+sineValue(model.spec.x)+' / '+sineKnobs[model.spec.y][0]+' '+sineValue(model.spec.y);
  $('#xyValue_'+name).textContent=text;model.el.setAttribute('aria-label',model.spec.title+'。'+text+'。矢印キーでも操作できます');
  $('#xyHold_'+name).classList.toggle('on',mode==='hold');$('#xyDrift_'+name).classList.toggle('on',mode==='drift');
  $('#xyHold_'+name).setAttribute('aria-pressed',String(mode==='hold'));$('#xyDrift_'+name).setAttribute('aria-pressed',String(mode==='drift'));
  $('#xyTrail_'+name).setAttribute('points',model.trail.map(p=>(p[0]*100).toFixed(2)+','+((1-p[1])*100).toFixed(2)).join(' '));
}
function sinePadAnchor(model){model.ax=model.x;model.ay=model.y;model.phase=0}
function sinePadTrace(model){model.trail.push([model.x,model.y]);if(model.trail.length>40)model.trail.shift()}
function sinePadPointer(model,e){const r=model.el.getBoundingClientRect();if(!r.width||!r.height)return;sinePadApply(model,(e.clientX-r.left)/r.width-(model.grabX||0),1-(e.clientY-r.top)/r.height-(model.grabY||0));sinePadTrace(model);sinePadRender(model)}
function sinePadSync(){
  for(const model of Object.values(sinePadModels)){if(model.pointer!==null)continue;model.x=sinePadNorm(model.spec.x);model.y=sinePadNorm(model.spec.y);model.trail=[];sinePadAnchor(model);sinePadRender(model)}
  $('#xyDriftRange').value=String(sineState.driftAmount);
}
const padsRoot=$('#sinePads');
padsRoot.innerHTML=`<div class="sinePadGrid">${Object.entries(sinePadSpecs).map(([name,spec])=>`<section class="sinePadCard"><div class="sinePadTitle"><strong>${spec.title}</strong><div><button id="xyHold_${name}" type="button">HOLD</button><button id="xyDrift_${name}" type="button">DRIFT</button></div></div><div class="sineXY" id="xy_${name}" role="group" tabindex="0"><svg viewBox="0 0 100 100" preserveAspectRatio="none" aria-hidden="true"><path d="M25 0V100 M50 0V100 M75 0V100 M0 25H100 M0 50H100 M0 75H100" class="xyGrid"/><polyline id="xyTrail_${name}" points="" class="xyTrail"/></svg><span class="xyAxis xyTop">${spec.top}</span><span class="xyAxis xyBottom">${spec.bottom}</span><span class="xyAxis xyLeft">${spec.left}</span><span class="xyAxis xyRight">${spec.right}</span><span class="xyDot" id="xyDot_${name}"></span></div><output class="xyValue" id="xyValue_${name}"></output></section>`).join('')}</div><div class="xyFooter"><span>光る点をつかんで音を操作 · 点以外はスクロール</span><label>漂う範囲 <select id="xyDriftRange"><option value="0.05">小さく</option><option value="0.15">ほどよく</option><option value="0.35">広く</option></select></label></div>`;
for(const [name,spec] of Object.entries(sinePadSpecs)){
  const model={name,spec,el:$('#xy_'+name),x:sinePadNorm(spec.x),y:sinePadNorm(spec.y),pointer:null,phase:0,trail:[]};sinePadModels[name]=model;sinePadAnchor(model);
  model.el.addEventListener('pointerdown',e=>{if(e.button!==0||model.pointer!==null||!e.target.closest('.xyDot'))return;e.preventDefault();model.el.focus();const r=model.el.getBoundingClientRect();model.grabX=(e.clientX-r.left)/r.width-model.x;model.grabY=1-(e.clientY-r.top)/r.height-model.y;model.pointer=e.pointerId;model.el.setPointerCapture(e.pointerId)});
  model.el.addEventListener('pointermove',e=>{if(model.pointer!==e.pointerId)return;e.preventDefault();sinePadPointer(model,e)});
  const release=e=>{if(model.pointer!==e.pointerId)return;model.pointer=null;sinePadAnchor(model);saveSine()};
  for(const type of ['pointerup','pointercancel','lostpointercapture'])model.el.addEventListener(type,release);
  model.el.addEventListener('keydown',e=>{const shifts={ArrowLeft:[-.025,0],ArrowRight:[.025,0],ArrowUp:[0,.025],ArrowDown:[0,-.025]};const shift=shifts[e.key];if(!shift)return;e.preventDefault();e.stopPropagation();const scale=e.shiftKey?4:1;sinePadApply(model,model.x+shift[0]*scale,model.y+shift[1]*scale);sinePadTrace(model);sinePadAnchor(model);sinePadRender(model);saveSine()});
  for(const mode of ['hold','drift'])$('#xy'+(mode==='hold'?'Hold':'Drift')+'_'+name).onclick=()=>{sineState[name+'Mode']=mode;sinePadAnchor(model);sinePadRender(model);saveSine()};
}
$('#xyDriftRange').onchange=e=>{sineState.driftAmount=clamp(Number(e.target.value),.05,.35);for(const model of Object.values(sinePadModels))sinePadAnchor(model);saveSine()};
let sinePadLast=0,sinePadSaved=0;
function sinePadTick(time){
  const dt=sinePadLast?clamp((time-sinePadLast)/1000,0,.05):0;sinePadLast=time;
  if(running&&sineState.enabled){for(const model of Object.values(sinePadModels)){if(model.pointer!==null||sineState[model.name+'Mode']!=='drift')continue;
    model.phase+=dt*.23;const p=model.phase,r=sineState.driftAmount;
    const x=model.ax+r*(Math.sin(p*1.1)*.65+Math.sin(p*2.3)*.35),y=model.ay+r*(Math.sin(p*.83)*.65+Math.sin(p*1.73)*.35);
    sinePadApply(model,x,y);if(time-sinePadSaved>150)sinePadTrace(model);
  }}
  if(time-sinePadSaved>1000){if(running&&sineState.enabled&&Object.values(sinePadModels).some(m=>sineState[m.name+'Mode']==='drift'))saveSine();sinePadSaved=time}
  requestAnimationFrame(sinePadTick);
}
window.sinePadController={sync:sinePadSync};sinePadSync();requestAnimationFrame(sinePadTick);
