function snapConfigValue(v,c){
  const clamped=clamp(v,c.min,c.max);
  if(!c.step)return clamped;
  const n=Math.round((clamped-c.min)/c.step);
  return clamp(c.min+n*c.step,c.min,c.max);
}
function makeRandomRangeBar(prefix,id,c,store,input,getValue,setValue){
  const shell=document.createElement('div');shell.className='rangeBarShell';shell.dataset.rangeKey=prefix+'_'+id;
  const windowEl=document.createElement('div');windowEl.className='rangeWindow';
  const minHandle=document.createElement('button');minHandle.type='button';minHandle.className='rangeBound rangeMin';minHandle.setAttribute('aria-label','Random minimum');
  const maxHandle=document.createElement('button');maxHandle.type='button';maxHandle.className='rangeBound rangeMax';maxHandle.setAttribute('aria-label','Random maximum');
  const labels=document.createElement('div');labels.className='rangeBoundLabels';
  const minLabel=document.createElement('span'),maxLabel=document.createElement('span');labels.append(minLabel,maxLabel);
  input.classList.add('rangeCurrent');
  shell.append(windowEl,input,minHandle,maxHandle,labels);

  const sync=()=>{
    const r=store[id];
    const nMin=clamp(toNorm(r.min,c),0,1),nMax=clamp(toNorm(r.max,c),0,1);
    windowEl.style.left=(nMin*100)+'%';
    windowEl.style.width=(Math.max(0,nMax-nMin)*100)+'%';
    minHandle.style.left=(nMin*100)+'%';
    maxHandle.style.left=(nMax*100)+'%';
    minLabel.textContent='MIN '+fmt(r.min,c);
    maxLabel.textContent='MAX '+fmt(r.max,c);
    minHandle.title='MIN '+fmt(r.min,c);
    maxHandle.title='MAX '+fmt(r.max,c);
  };
  const changeBound=(which,e)=>{
    e.preventDefault();e.stopPropagation();
    const handle=which==='min'?minHandle:maxHandle;
    try{handle.setPointerCapture(e.pointerId)}catch(_){}
    const move=ev=>{
      const rect=shell.getBoundingClientRect();
      const n=clamp((ev.clientX-rect.left)/Math.max(1,rect.width),0,1);
      let v=snapConfigValue(fromNorm(n,c),c);
      const r=store[id];
      if(which==='min')v=Math.min(v,r.max);else v=Math.max(v,r.min);
      r[which]=v;
      const cur=getValue();
      if(cur<r.min)setValue(r.min);
      else if(cur>r.max)setValue(r.max);
      sync();
    };
    const done=ev=>{
      handle.removeEventListener('pointermove',move);
      handle.removeEventListener('pointerup',done);
      handle.removeEventListener('pointercancel',done);
      try{handle.releasePointerCapture(ev.pointerId)}catch(_){}
      resetWalkAgents();
    };
    handle.addEventListener('pointermove',move);
    handle.addEventListener('pointerup',done);
    handle.addEventListener('pointercancel',done);
    move(e);
  };
  minHandle.addEventListener('pointerdown',e=>changeBound('min',e));
  maxHandle.addEventListener('pointerdown',e=>changeBound('max',e));
  shell._syncRange=sync;
  sync();
  return shell;
}

}
