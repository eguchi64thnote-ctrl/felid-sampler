function makeDefault1Snapshot(){
  const valuesFrom=defs=>Object.fromEntries(Object.entries(defs).map(([id,c])=>[id,c.def]));
  const rangesFrom=defs=>Object.fromEntries(Object.entries(defs).map(([id,c])=>[id,{min:c.rmin??c.min,max:c.rmax??c.max}]));
  return {version:1,savedAt:0,state:valuesFrom(configs),meta:valuesFrom(metaConfigs),rhythmState:valuesFrom(rhythmConfigs),mixerState:valuesFrom(mixerConfigs),randomRanges:rangesFrom(configs),rhythmRandomRanges:rangesFrom(rhythmConfigs),mixerRandomRanges:rangesFrom(mixerConfigs),holds:{ambient:[],rhythm:[],mixer:['e808kick','bassAttack','bassRelease']},randomWalk:true,rhythm:{on:false,grooveFocus:'auto',groovePrimary:'house',grooveSecondary:'jazz',grooveBlendAmount:.05,divisionFocus:'auto',currentDivision:11},groups:[...GROUPS.values()].map(g=>({id:g.id,active:g.builtin?true:false,weight:g.builtin?1:(Number(g.weight)||1)})),materials:[...MATERIALS.values()].map(m=>({id:m.id,enabled:m.builtin?true:false,groupId:m.groupId}))};
}
function randomizeDefaultSampleSelection(){
  const mats=[...MATERIALS.values()];
  if(!mats.length)return;
  const count=Math.min(mats.length,1+Math.floor(rand()*3));
  const pool=[...mats];
  const chosen=new Set();
  while(chosen.size<count&&pool.length){
    const i=Math.floor(rand()*pool.length);
    const [m]=pool.splice(i,1);
    if(m)chosen.add(m.id);
  }
  for(const m of mats)m.enabled=chosen.has(m.id);
  const activeGroups=new Set(mats.filter(m=>chosen.has(m.id)).map(m=>m.groupId));
  for(const g of GROUPS.values())g.active=activeGroups.has(g.id);
  renderGroups();
}

    let timer=null,longFired=false;const slot=b.dataset.slot;
    b.addEventListener('pointerdown',()=>{longFired=false;if(snapshotSlots[slot])timer=setTimeout(()=>{longFired=true;saveSnapshot(slot);if(navigator.vibrate)navigator.vibrate(25)},650)});
    const finish=()=>{if(timer){clearTimeout(timer);timer=null}if(longFired){longFired=false;return}if(snapshotSlots[slot])restoreSnapshot(slot);else saveSnapshot(slot)};
    b.addEventListener('pointerup',finish);b.addEventListener('pointercancel',()=>{if(timer)clearTimeout(timer);timer=null});b.addEventListener('pointerleave',()=>{if(timer)clearTimeout(timer);timer=null});b.addEventListener('contextmenu',e=>e.preventDefault());
    b.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();if(e.shiftKey)saveSnapshot(slot);else if(snapshotSlots[slot])restoreSnapshot(slot);else saveSnapshot(slot)}});
  });
  persistMemoryState();updateSnapshotButtons();
}
