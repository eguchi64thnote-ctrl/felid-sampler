function drawKaresansui(w,h,low,mid,high,t){
  cc.fillStyle='rgba(7,7,7,'+(0.18+(1-visualState.trail)*.58)+')';cc.fillRect(0,0,w,h);
  const intensity=visualState.intensity*(.22+(low+mid+high)/3*visualState.reactivity);
  cc.save();cc.translate(w/2,h/2);cc.scale(visualState.zoom,visualState.zoom);cc.translate(-w/2,-h/2);
  const lines=10+Math.round(visualState.particles*15);
  for(let j=0;j<lines;j++){
    const y0=(j+1)*h/(lines+1),phase=t*(.12+.18*visualState.motion)+j*.62;
    cc.beginPath();
    for(let x=-20;x<=w+20;x+=5){
      const n=clamp(x/w,0,1),spec=spectrumAt(n);
      const wave=Math.sin(n*Math.PI*(3+j*.12)+phase)*h*.012*visualState.motion;
      const bend=Math.sin(n*Math.PI*2+phase*.43+j)*h*.008;
      const audio=(spec*spec)*h*.11*intensity*(.35+Math.sin(j*.71)*.22);
      const y=y0+wave+bend-audio;

  }
  const rings=2+Math.floor(low*5*visualState.reactivity);

  }
